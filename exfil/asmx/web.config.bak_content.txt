    <script type="text/javascript">
        function validateLogin() {
            if ($('#txtInstitutionCode').val().trim() == '') {
                validateError('txtInstitutionCode', 'Please enter Institution Code');
                return false;
            }
            else {
                validateSuccess('txtInstitutionCode');
            }
            return true;

        }

        function validateLoginBylink() {
            $('#Button1').css('display', 'inline-block');
            if ($('#txtWebsiteUrl').val().trim() == '') {
                validateError('txtWebsiteUrl', 'Please enter Institution website link');
                return false;
            }
            else {
                validateSuccess('txtWebsiteUrl');
            }
            return true;
        }

        function OnKeyPress(e) {
            debugger;
            var keycode = (e.which) ? e.which : e.keyCode;
            if (e.which == 13) {
                if (validateLogin()) {

                    var btn = document.getElementById('btn_Login');
                    btn.click();
                }
                else { return false; }
            }
            else if ((keycode > 64 && keycode < 91) || (keycode > 96 && keycode < 123)) {
                return true;
            }
            else {
                return false;
            }
        }

        function OnKeyPress1(e) {
            debugger;
            if (e.which == 13) {
                if (validateLoginBylink()) {
                    var btn = document.getElementById('Button1');
                    btn.click();
                }
                else { return false; }
            }
        }
    </script>
